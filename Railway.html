public static List<Train> quickSort(List<Train> list, Comparator<Train> comparator, SortMetrics metrics) {
    if (list == null || list.size() <= 1) return list;
    long start = System.nanoTime();
    quickSortHelper(list, 0, list.size() - 1, comparator, metrics);
    metrics.executionTimeNanos = System.nanoTime() - start;
    return list;
}

private static int partition(List<Train> list, int low, int high, Comparator<Train> comparator, SortMetrics metrics) {
    // Median-of-three pivot selection
    int mid = low + (high - low) / 2;
    if (comparator.compare(list.get(mid), list.get(low)) < 0) swap(list, low, mid, metrics);
    if (comparator.compare(list.get(high), list.get(low)) < 0) swap(list, low, high, metrics);
    if (comparator.compare(list.get(high), list.get(mid)) < 0) swap(list, mid, high, metrics);

    Train pivot = list.get(high);
    int i = low - 1;
    for (int j = low; j < high; j++) {
        metrics.comparisons++;
        if (comparator.compare(list.get(j), pivot) <= 0) {
            i++;
            swap(list, i, j, metrics);
        }
    }
    swap(list, i + 1, high, metrics);
    return i + 1;
}
